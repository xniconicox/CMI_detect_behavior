# クラス別性能分析
print("クラス別性能を分析中...")

# ラベルエンコーダーでクラス名を取得
if hasattr(preprocessor, 'label_encoder') and preprocessor.label_encoder is not None:
    class_names = preprocessor.label_encoder.classes_
else:
    class_names = [f'Class_{i}' for i in range(len(np.unique(y_true)))]

# 各クラスについて個別モデルとアンサンブルの性能を比較
class_results = []

for class_idx in range(len(class_names)):
    class_mask = (y_true == class_idx)
    if class_mask.sum() == 0:
        continue
        
    # 個別モデルの性能
    fold_f1_scores = []
    for fold in range(N_FOLDS):
        y_pred_fold = fold_predictions[fold][class_mask]
        y_true_class = y_true[class_mask]
        
        # このクラスが存在する場合のみF1スコアを計算
        if len(np.unique(y_true_class)) > 1:  # 複数クラスが存在する場合
            # このクラスを正例、他を負例としてbinary F1を計算
            y_true_binary = (y_true_class == class_idx).astype(int)
            y_pred_binary = (y_pred_fold == class_idx).astype(int)
            
            # 正例が存在する場合のみF1スコアを計算
            if np.sum(y_true_binary) > 0:
                f1 = f1_score(y_true_binary, y_pred_binary, average='binary', zero_division=0)
            else:
                f1 = 0.0
        else:
            # 単一クラスの場合
            f1 = 1.0 if y_pred_fold[0] == y_true_class[0] else 0.0
        
        fold_f1_scores.append(f1)
    
    # アンサンブルの性能
    y_pred_ensemble = ensemble_predictions[class_mask]
    y_true_class = y_true[class_mask]
    
    if len(np.unique(y_true_class)) > 1:
        y_true_binary = (y_true_class == class_idx).astype(int)
        y_pred_binary = (y_pred_ensemble == class_idx).astype(int)
        
        if np.sum(y_true_binary) > 0:
            ensemble_f1 = f1_score(y_true_binary, y_pred_binary, average='binary', zero_division=0)
        else:
            ensemble_f1 = 0.0
    else:
        ensemble_f1 = 1.0 if y_pred_ensemble[0] == y_true_class[0] else 0.0
    
    class_results.append({
        'class_name': class_names[class_idx],
        'class_idx': class_idx,
        'sample_count': class_mask.sum(),
        'fold_f1_mean': np.mean(fold_f1_scores),
        'fold_f1_std': np.std(fold_f1_scores),
        'ensemble_f1': ensemble_f1,
        'improvement': ensemble_f1 - np.mean(fold_f1_scores)
    })
